# Packages this setup needs, installed by install.sh with
#   sudo pacman -S --needed
# Blank lines and #-comments are ignored. All are in the official repos.

# compositor
hyprland

# the apps configured under config/
foot
fuzzel
mako
hyprlock
hypridle
waybar
neovim
herdr

# browsers. zen is the daily one; chromium is for Electron-style web apps
# (Slack, Spotify) so they don't each ship their own runtime.
# zen-browser-bin is AUR, like herdr — see the yay note in install.sh.
zen-browser-bin
chromium

# runtime version manager — config.toml pins node/bun/zig/etc per project
mise

# API client (AUR) and the game engine
bruno-bin
godot

# file transfer to phones/laptops on the LAN (AUR; the non-bin one builds
# Flutter from source, which takes an age)
localsend-bin

# viewers. zathura needs a backend to open anything — mupdf is the light one.
# imv is configured under config/imv; mpv is used as shipped.
zathura
zathura-pdf-mupdf
imv
mpv

# imv's Ctrl+r bind shells out to mogrify, and its Ctrl+p bind to lp
imagemagick

# printing. cups is the daemon (install.sh enables it), system-config-printer
# is the GUI that "print settings" actually means.
cups
system-config-printer

# lstopo — hardware topology
hwloc

# remote-viewer, for SPICE/VNC guests
virt-viewer

# what nvim and herdr actually shell out to. LazyVim's pickers are ripgrep and
# fd; herdr's prefix+n / prefix+shift+g popups are yazi and lazygit; unzip is
# how Mason unpacks language servers.
ripgrep
fd
lazygit
yazi
unzip

# wallpaper — a solid colour, so no image asset is needed
swaybg

# screenshots. also xdg-desktop-portal-hyprland's optional deps: without them
# the screenshot/screencast portal doesn't work for other apps either.
grim
slurp

# laptop function keys (volume goes through wpctl, from wireplumber)
brightnessctl

# notify-send, used by the SUPER+B status bind. mako does NOT pull this in.
libnotify

# Qt apps run native Wayland instead of falling back to XWayland
qt6-wayland

# night light. 0.4.0 has no config file — it's driven entirely by CLI flags,
# so it's a toggle bind rather than a scheduled daemon.
hyprsunset

# clipboard history, picked with fuzzel
cliphist

# audio beyond the volume keys: routing GUI and media-key control
pavucontrol
playerctl

# network and bluetooth. Both are daemons — install.sh enables them, since
# installing them without starting them leaves you with no wifi.
# iwd rather than NetworkManager: impala is an iwd TUI (`Depends On: iwd`), and
# the two are competing wifi backends — running both makes them fight over the
# radio. iwd does wifi; addresses and DNS come from systemd-networkd/resolved,
# which are already in `base` via systemd.
# impala and bluetui are keyboard-driven TUIs (bluetui is the in-repo
# equivalent of bluetuith, which is AUR-only).
iwd
impala
bluez
bluez-utils
bluetui

# font referenced by foot.ini and fuzzel.ini, plus emoji so notifications and
# chat apps don't render tofu. noto-fonts is the proportional face: the nerd
# font is mono-only, so without it GTK apps and waybar tooltips have nothing
# to fall back to on a bare install.
ttf-jetbrains-mono-nerd
noto-fonts
noto-fonts-emoji

# creates ~/Pictures and friends, where the Print screenshot bind writes
xdg-user-dirs

# clipboard, screen sharing / file pickers, GUI authentication prompts
wl-clipboard
xdg-desktop-portal-hyprland
xdg-desktop-portal-gtk
hyprpolkitagent

# audio — nothing here configures it, but a desktop without sound is no fun.
# pipewire-alsa gives ALSA-only apps a route in; without it they find no device.
pipewire
pipewire-pulse
pipewire-alsa
wireplumber
